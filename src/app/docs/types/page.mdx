## Base Types

### EntityId

API responses expose `entityId` as a numeric identifier.

```typescript
type EntityId = number;
```

### Minimal Relationship Types

Related entities use minimal projections by default.

```typescript
type PersonMinimal = {
  entityId: number;
  id: string;
  name: string;
};

type FilmMinimal = {
  entityId: number;
  id: string;
  title: string;
  episode: number;
};

type PlanetMinimal = {
  entityId: number;
  id: string;
  name: string;
};

type SpeciesMinimal = {
  entityId: number;
  id: string;
  name: string;
};

type StarshipMinimal = {
  entityId: number;
  id: string;
  name: string;
};

type VehicleMinimal = {
  entityId: number;

};
```

## Usage Examples

### Fetch with Types

```typescript
async function getPerson(id: string): Promise<PersonResponse> {
  const response = await fetch(
    `https://sw-next-api.vercel.app/api/v1/people/${id}`
  );

  if (!response.ok) {
    throw new Error("Failed to fetch person");
  }

  return response.json();
}

const luke = await getPerson("luke-skywalker");
console.log(luke.name);
console.log(luke.heightCm);
console.log(luke.meta.isJedi);
```

### Paginated Fetch

```typescript
async function getPeople(
  page: number = 1,
  limit: number = 10
): Promise<PaginatedResponse<PersonResponse>> {
  const params = new URLSearchParams({
    page: page.toString(),
    limit: limit.toString(),
  });

  const response = await fetch(
    `https://sw-next-api.vercel.app/api/v1/people?${params}`
  );

  return response.json();
}

const { results, total, pages } = await getPeople(1, 20);
console.log(`Showing ${results.length} of ${total} people across ${pages} pages`);
```

### Working with `expand`

By default, related entities use minimal projections. When you request
`expand`, the response shape becomes richer for those specific fields.

```typescript
type ExpandedPersonResponse = Omit<PersonResponse, "homeworld" | "films"> & {
  homeworld: PlanetResponse | null;
  films: FilmResponse[];
};

async function getPersonExpanded(id: string): Promise<ExpandedPersonResponse> {
  const response = await fetch(
    `https://sw-next-api.vercel.app/api/v1/people/${id}?expand=homeworld,films`
  );

  return response.json();
}

const luke = await getPersonExpanded("luke-skywalker");
console.log(luke.homeworld?.climate);
console.log(luke.films[0]?.director);
```

## Type Guards

Useful type guards for runtime checks:

```typescript
function isPersonResponse(entity: unknown): entity is PersonResponse {
  return (
    typeof entity === "object" &&
    entity !== null &&
    "entityId" in entity &&
    "id" in entity &&
    "name" in entity &&
    "meta" in entity
  );
}

function isMinimalPerson(entity: unknown): entity is PersonMinimal {
  return (
    typeof entity === "object" &&
    entity !== null &&
    "entityId" in entity &&
    "id" in entity &&
    "name" in entity &&
    Object.keys(entity).length === 3
  );
}
```
