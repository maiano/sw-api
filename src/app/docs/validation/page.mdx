## Pagination Schema

```typescript
const PaginatedResponseSchema = <T extends z.ZodType>(itemSchema: T) =>
  z.object({
    page: z.number().int().positive(),
    limit: z.number().int().positive(),
    total: z.number().int().nonnegative(),
    pages: z.number().int().nonnegative(),
    results: z.array(itemSchema),
  });

const PeoplePaginatedSchema = PaginatedResponseSchema(PersonSchema);
const FilmsPaginatedSchema = PaginatedResponseSchema(FilmSchema);

type PeoplePaginated = z.infer<typeof PeoplePaginatedSchema>;
type FilmsPaginated = z.infer<typeof FilmsPaginatedSchema>;
```

## Validation Examples

### Basic Validation

```typescript
async function getPerson(id: string): Promise<PersonResponse> {
  const response = await fetch(
    `https://sw-next-api.vercel.app/api/v1/people/${id}`
  );

  const data = await response.json();

  return PersonSchema.parse(data);
}

try {
  const luke = await getPerson("luke-skywalker");
  console.log(luke.name);
} catch (error) {
  if (error instanceof z.ZodError) {
    console.error("Validation failed:", error.issues);
  }
}
```

### Safe Parsing

```typescript
async function getPersonSafe(id: string) {
  const response = await fetch(
    `https://sw-next-api.vercel.app/api/v1/people/${id}`
  );

  const data = await response.json();
  const result = PersonSchema.safeParse(data);

  if (!result.success) {
    console.error("Validation errors:", result.error.issues);
    return null;
  }

  return result.data;
}

const luke = await getPersonSafe("luke-skywalker");
if (luke) {
  console.log(luke.name);
}
```

### Expanded Responses

When you use `expand`, validate the richer shape explicitly instead of
re-using the default minimal relationship schema.

```typescript

      id: z.string(),
      name: z.string(),
      climate: z.string(),
      gravity: z.string(),
      terrain: z.string(),

      episode: z.number(),
      director: z.string(),
      release_date: z.string(),
      release_year: z.number(),

  ),
```

## Client-Side Validation

### React Hook

```typescript
import { useEffect, useState } from "react";
import { z } from "zod";

function usePerson(id: string) {
  const [person, setPerson] = useState<PersonResponse | null>(null);
  const [error, setError] = useState<string | null>(null);
  const [loading, setLoading] = useState(false);

  useEffect(() => {
    async function fetchPerson() {
      setLoading(true);
      setError(null);

      try {
        const response = await fetch(
          `https://sw-next-api.vercel.app/api/v1/people/${id}`
        );

        const data = await response.json();
        const validated = PersonSchema.parse(data);

        setPerson(validated);
      } catch (err) {
        if (err instanceof z.ZodError) {
          setError("Invalid data received from API");
          console.error("Validation errors:", err.issues);
        } else {
          setError("Failed to fetch person");
        }
      } finally {
        setLoading(false);
      }
    }

    fetchPerson();
  }, [id]);

  return { person, error, loading };
}

function PersonProfile({ id }: { id: string }) {
  const { person, error, loading } = usePerson(id);

  if (loading) return <div>Loading...</div>;
  if (error) return <div>Error: {error}</div>;
  if (!person) return null;

  return (
    <div>
      <h1>{person.name}</h1>
      <p>Height: {person.heightCm ?? "unknown"} cm</p>
      <p>Homeworld: {person.homeworld?.name ?? "unknown"}</p>
    </div>
  );
}
```

### Form Validation

```typescript
const SearchFormSchema = z.object({
  search: z.string().min(1, "Search query is required"),
  gender: z.enum(["male", "female", "n/a", "none"]).optional(),
  faction: z
    .enum([
      "republic",
      "separatists",
      "empire",
      "rebels",
      "civilian",
      "unknown",
    ])
    .optional(),
  page: z.coerce.number().int().positive().default(1),
  limit: z.coerce.number().int().min(1).max(100).default(10),
});

type SearchFormData = z.infer<typeof SearchFormSchema>;

function handleSearch(formData: unknown) {
  const result = SearchFormSchema.safeParse(formData);

  if (!result.success) {
    result.error.issues.forEach((issue) => {
      console.error(`${issue.path.join(".")}: ${issue.message}`);
    });
    return;
  }

  const { search, gender, faction, page, limit } = result.data;
  console.log({ search, gender, faction, page, limit });
}
```

## Server-Side Validation

### API Route Handler

```typescript
import { NextRequest, NextResponse } from "next/server";
import { z } from "zod";

const QueryParamsSchema = z.object({
  page: z.coerce.number().int().positive().default(1),
  limit: z.coerce.number().int().min(1).max(100).default(10),
  search: z.string().optional(),
  gender: z.enum(["male", "female", "n/a", "none"]).optional(),
  faction: z
    .enum([
      "republic",
      "separatists",
      "empire",
      "rebels",
      "civilian",
      "unknown",
    ])
    .optional(),
  sort: z.string().optional(),
  expand: z.string().optional(),
});

export async function GET(request: NextRequest) {
  const { searchParams } = new URL(request.url);
  const params = Object.fromEntries(searchParams.entries());
  const result = QueryParamsSchema.safeParse(params);

  if (!result.success) {
    return NextResponse.json(
      {
        error: "Invalid query parameters",
        details: result.error.issues,
      },
      { status: 400 }
    );
  }

  const { page, limit, search, gender, faction, sort, expand } = result.data;

  return NextResponse.json({
    page,
    limit,
    search,
    gender,
    faction,
    sort,
    expand,
  });
}
```

## Error Handling

### Formatted Error Messages

```typescript
import { z } from "zod";

function formatZodError(error: z.ZodError): string[] {
  return error.issues.map((issue) => {
    const path = issue.path.join(".");
    return `${path}: ${issue.message}`;
  });
}

try {
  PersonSchema.parse(data);
} catch (error) {
  if (error instanceof z.ZodError) {
    const messages = formatZodError(error);
    console.error("Validation errors:", messages);
  }
}
```

### Custom Error Messages

```typescript
const PersonSchema = z.object({
  entityId: z.number({
    error: "Entity ID must be a number",
  }),
  name: z.string().min(1, "Name cannot be empty"),
  heightCm: z
    .number()
    .nullable()
    .refine((value) => value === null || value > 0, "Height must be positive"),
});
```
